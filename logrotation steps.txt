# ===============================
# Command Logging to Syslog
# ===============================

# Global variable to track last logged command
function log2syslog_custom {
    local COMMAND TERMINAL

    case "$BASH_COMMAND" in
        history*|logger*|"trap"*|"") return ;;
    esac

    COMMAND=$(fc -ln -0 2>/dev/null)
    TERMINAL=$(tty 2>/dev/null)
    [[ $TERMINAL == "not a tty" || -z $TERMINAL ]] && TERMINAL=${SSH_TTY:-unknown}
    TERMINAL=$(echo "$TERMINAL" | awk -F/ '{print $(NF-1) "/" $NF}')

    [ -n "$COMMAND" ] && logger -p local1.notice -t bash -- "${TERMINAL}:${USER}:${COMMAND}"
}
